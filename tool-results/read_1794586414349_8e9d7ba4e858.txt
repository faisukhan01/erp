   164→function GenerationBubble() {
   165→  const { genStage, streamedCode, genStartedAt } = useStore();
   166→  const [elapsed, setElapsed] = useState(0);
   167→
   168→  useEffect(() => {
   169→    if (!genStartedAt) return;
   170→    const t = setInterval(() => setElapsed(Math.round((Date.now() - genStartedAt) / 1000)), 1000);
   171→    return () => clearInterval(t);
   172→  }, [genStartedAt]);
   173→
   174→  const current = stageIndex(genStage);
   175→
   176→  return (
   177→    <motion.div
   178→      initial={{ opacity: 0, y: 8 }}
   179→      animate={{ opacity: 1, y: 0 }}
   180→      exit={{ opacity: 0 }}
   181→      className="flex justify-start"
   182→    >
   183→      <div className="w-full max-w-[92%] rounded-2xl rounded-bl-md border border-rose-500/20 bg-rose-500/[0.05] px-3.5 py-3">
   184→        {/* stepper */}
   185→        <div className="flex items-center gap-1.5 flex-wrap">
   186→          {STAGES.map((s, i) => {
   187→            const done = i < current;
   188→            const active = i === current;
   189→            return (
   190→              <div
   191→                key={s.key}
   192→                className={`flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-[10px] font-medium transition-colors ${
   193→                  done
   194→                    ? "border-white/15 bg-white/[0.06] text-zinc-300"
   195→                    : active
   196→                      ? "animate-pulse-ring border-rose-500/40 bg-rose-500/15 text-rose-200"
   197→                      : "border-white/[0.06] bg-white/[0.02] text-zinc-600"
   198→                }`}
   199→              >
   200→                {done ? <Check className="h-3 w-3" /> : <s.icon className={`h-3 w-3 ${active ? "animate-pulse" : ""}`} />}
   201→                {s.label}
   202→              </div>
   203→            );
   204→          })}
   205→          <span className="ml-auto flex items-center gap-1 text-[10px] text-zinc-500 tabular-nums">
   206→            <Loader2 className="h-3 w-3 animate-spin text-rose-400" />
   207→            {elapsed}s
   208→          </span>
   209→        </div>
   210→
   211→        {/* live code counter */}
   212→        <div className="mt-2.5 flex items-center gap-2 text-[11px] text-zinc-500">
   213→          <span className="inline-block h-1.5 w-1.5 rounded-full bg-rose-400 animate-pulse" />
   214→          <span className="tabular-nums">{streamedCode.length.toLocaleString()}</span> characters written
   215→          <span className="text-zinc-600">— Z.AI is typing…</span>
   216→        </div>
   217→
   218→        {/* mini progress shimmer */}
   219→        <div className="relative mt-2.5 h-1 overflow-hidden rounded-full bg-white/[0.05]" aria-hidden>
   220→          <div className="animate-stream-line absolute inset-y-0 w-1/3 rounded-full bg-gradient-to-r from-transparent via-rose-400/80 to-transparent" />
   221→        </div>
   222→      </div>
   223→    </motion.div>
   224→  );
   225→}
